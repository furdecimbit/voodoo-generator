# Pack Room Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Rebuild `pack-room.html` as a standalone 3-panel pack management room — wallet connect + pack inventory (left), pack opening + item reveal (center), session stats (right).

**Architecture:** Single self-contained HTML file. Left panel handles wallet (Solana + Tezos) and renders owned pack cards fetched via Helius. Center panel runs the pack opening flow: pack floats → click to open → video → item cards appear one by one, each clicked to reveal. Right panel shows session open count by rarity. All wallet/Helius/Wix logic copied from `pack_test.html`.

**Tech Stack:** Vanilla JS, CSS Grid (3-column), Bloodcrow font via CDN, Helius DAS API, Wix HTTP functions, Phantom/Solflare/Backpack (Solana), Beacon (Tezos).

```js
const PACK_IMGS = { common:'common_pack.png', rare:'rare_pack.png', legendary:'legendary_pack.png', ultimate:'ultimate_pack.png' };
const RARITIES = ['ultimate','legendary','rare','common'];

function renderPackList(counts, mints){
  const list = document.getElementById('pack-list');
  document.getElementById('packs-empty-msg').style.display = 'none';
  list.innerHTML = '';
  let any = false;
  for(const r of RARITIES){
    if(!counts[r]) continue;
    any = true;
    const entry = document.createElement('div');
    entry.className = 'pack-entry';
    entry.dataset.rarity = r;
    entry.innerHTML = `
      <img src="${PACK_IMGS[r]}" alt="${r}">
      <div class="pack-entry-info">
        <div class="pack-entry-name rarity-${r}">${r} Pack</div>
        <div class="pack-entry-count">${counts[r]} unopened</div>
      </div>`;
    entry.onclick = () => selectPack(r, mints[r]?.[0]);
    list.appendChild(entry);
  }
  if(!any){
    document.getElementById('packs-empty-msg').style.display = '';
    document.getElementById('packs-empty-msg').textContent = 'No unopened packs';
  }
}
```

let activePack = null; // { rarity, mint }

function selectPack(rarity, mint){
  activePack = { rarity, mint };

  document.getElementById('center-reveal').style.display = 'none';

  title.className = `rarity-${rarity}`;

  document.getElementById('pack-img').src = PACK_IMGS[rarity];

```js
function collectPack(){
  const { rarity, mint } = activePack;
  // Mark mint opened in localStorage
  if(mint && currentWallet){
    const key = `${currentWallet}_openedMints`;
    const opened = JSON.parse(localStorage.getItem(key) || '[]');
    if(!opened.includes(mint)) opened.push(mint);
    localStorage.setItem(key, JSON.stringify(opened));
  }

  // Decrement pack count and remove used mint
  packCounts[rarity] = Math.max(0, (packCounts[rarity] || 1) - 1);
  if(window._packMints?.[rarity]?.length){
    window._packMints[rarity].shift();
  }

  // Log to right panel
  logPackOpen(rarity);

  // Re-render left panel list
  renderPackList(packCounts, window._packMints);

  // Reset center
  activePack = null;
  document.getElementById('center-reveal').style.display = 'none';
  document.getElementById('cards-row').innerHTML = '';
  document.getElementById('collect-btn').style.display = 'none';
  document.getElementById('tap-hint').style.display = '';
  
  if(Object.values(packCounts).some(v => v > 0)){
    document.getElementById('center-idle').style.display = '';
  } else {
    document.getElementById('center-idle').style.display = '';
    document.getElementById('center-idle').innerHTML = 
      `<div style="font-size:14px;color:#3a2808;letter-spacing:.12em;text-transform:uppercase;text-align:center">All packs opened!</div>`;
  }
}
```
